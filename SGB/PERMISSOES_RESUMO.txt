  {
      if (SessaoUsuario.Perfil != "Bibliotecario" &&
          SessaoUsuario.Perfil != "Administrador")
      {
          MessageBox.Show("Você não possui permissão...", "Acesso negado");
          Close();
          return;
      }
      
      // Só chega aqui se tiver permissão
      CarregarUsuarios();
      CarregarEspacos();
      CarregarReservas();
  }
